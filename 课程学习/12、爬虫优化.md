```
# 导入同一个文件夹的时候，如果实在导入不出来，可以重启下idea，可能是缓存问题，然后在导入的前面加一个.表示从当前目录去查询  
# 类，在函数的基础上，还有一层，比他高级的函数，叫做类，他使用来包裹函数，这样子可以直接通过这个类来调用他下面的函数  
# 如果类，定义了__init__函数，那么他就是一个动态类，他就支持外部传输一个变量进来，作为他的类变量，这样他下面的函数都能去获取到这些类变量  
# 其他函数之间要能互相获取到，或者获取到类的全局变量，那么他的参数中，必须要加上self  
  
  
# python内置的库，无需去安装  
import json  
from writefile import writeFile  
  
  
# 静态类  
  
class getDocUtil:  
  
    def __init__(self, url, loginPayload, session, path):  
        # 我在这里定义了一个全局变量叫url，然后我将这个传递的url赋值给他  
        self.url = url  
        self.loginPayload = loginPayload  
        self.session = session  
        self.path = path  
        pass  
  
    # 这是请求实际数据接口  
    def getDocInfo(self, list, auth):  
  
        url = self.url + "/documentMenu/find/dataInfo"  
  
        heads = {  
            "authorization": auth  
        }  
  
        for i in list:  
            # {id:id,name:name}  
            payload = {"menuId": str(i["id"])}  
            re = self.session.post(url, headers=heads, json=payload, timeout=2, proxies={ "http" : None , "https" : None })  
  
            re = json.loads(re.text)  
            if re["code"] != 200:  
                print("这次请求失败")  
                print(re)  
                continue  
  
            if not re["data"]:  
                continue  
  
            text = re["data"]["textInfo"]  
  
            fileName = i["name"]  
            fileName = fileName.replace("/", "-")  
  
            writeFile(path=self.path, fileName=(fileName + ".html"), text=text, isWrite=True)  
  
    def getMenu(self, auth):  
        url = self.url + "/documentMenu/find/listMenu"  
        payload = {  
            "pageSize": 30,  
            "pageNumber": 1  
        }  
  
        if auth == None:  
            print("这里没有获取到jwt，停止执行")  
            return None  
  
        heads = {  
            "authorization": auth  
        }  
  
        re = self.session.post(url, json=payload, headers=heads, timeout=2, proxies={ "http" : None , "https" : None })  
  
        # 状态码，请求的状态信息，只有200，就是成功，如果是302，说明是要浏览器进行跳转，如果是其他非200或者302的，都是错误  
        if re.status_code != 200:  
            print("访问服务端失败")  
            return None  
        # re.content,用来获取二进制流，text获取文本信息  
        return self.checkData(json.loads(re.text))  
  
    # 清洗menuId数据  
    def checkData(self, res):  
        # 因为if这里的条件，会被强行进行一个bool值转换，所以我们可以利用这个特性，直接传递参数，由于这里为None的时候，会返回False,这样if条件就不执行了，所以我们加一个not，取反，如果是None,那么就执行这个if条件  
        if not res:  
            print("前面的流程失败了，无法清洗数据")  
            return None  
  
        if res["code"] != 200:  
            print("返回的数据，服务器端认为是错误的请求")  
            return None  
  
        # 这里大家不要懵逼，因为这里data下面还有一个data  
        data = res["data"]["data"]  
  
        list = []  
        for i in data:  
            list.append({"id": i["id"], "name": i["name"]})  
            sonList = i["list"]  
            if not sonList:  
                continue  
            for j in sonList:  
                list.append({"id": j["id"], "name": j["name"]})  
                sonSonList = j["list"]  
  
                if not sonSonList:  
                    continue  
  
                for k in sonSonList:  
                    list.append({"id": k["id"], "name": k["name"]})  
  
        return list  
  
    def login(self):  
  
        url = self.url + "/common/login"  
  
        print(self.loginPayload)  
        print(type(self.loginPayload))  
        print(url)  
  
        re = self.session.post(url, json=self.loginPayload, timeout=2, proxies={ "http" : None , "https" : None })  
  
        if re.status_code != 200:  
            print("请求登录接口，访问失败")  
            return None  
  
        # json提供了load和loads,大家不要搞混了，json处理文件信息，你就需要去调用load(),这里处理字符串信息，使用loads()去处理  
        dist = json.loads(re.text)  
  
        if dist["code"] != 200:  
            print("请求成功了，但是我们访问登录接口，可能是携带信息不对，导致服务端不认可我们的请求，返回了错误的信息")  
            print(re.text)  
            return None  
  
        data = dist["data"]  
        return data  
  
    def getData(self):  
        auth = self.login()  
        menuData = self.getMenu(auth)  
        self.getDocInfo(auth=auth, list=menuData)
```

```
def writeFile(path,fileName,text,isWrite=True):  
    w = "a"  
    if isWrite:  
        w = "w"  
    with open(path + "/" + fileName,w,encoding="utf-8") as f:  
        f.write(text)
```
