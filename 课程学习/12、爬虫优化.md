import requests   # 导入库
  
  
# 在python代码中，都会定义一个规范，大家会把执行代码都放到main  
  
def main():   
    while True:  
        url = input("请输入要爬取的网站地址：")  
  
        path = input("保存位置给一个：")  
        payload = {  
            "name": "961560092@qq.com",  
            "password": "123456"  
        }  
  
        with requests.session() as session:  
            get = getDocUtil(url, payload, session, path)  
  
            get.getData()  
  
  
# 一个判断语句，判断是不是在执行当前这个文件，如果是，就调用main()方法  
  
  
if __name__ == "__main__":  
    main()
