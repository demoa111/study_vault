# 1、当前网站获取数据需要两个接口，先获取菜单，然后使用菜单中的id获取出真实的数据 
# 2、先爬菜单  
# 3、清洗爬出的数据  
# 4、通过菜单的id找到真实数据  
# 5、将获取的数据保存到本地

import requests  
import json  
  
def login():  
    session = requests.session()  
    url = "http://47.108.197.167:8008/common/login"  
    payload = {  
        "name": "961560092@qq.com",  
        "password": "123456"  
    }  
  
    re = session.post(url, json=payload, timeout=2)  
    if re.status_code != 200:  
        print("请求登录接口，访问失败")  
        return None  
  
    # json提供了load和loads,大家不要搞混了，json处理文件信息，你就需要去调用load(),这里处理字符串信息，使用loads()去处理  
    dist = json.loads(re.text)  
  
    if dist["code"] != 200:  
        print("请求成功了，但是我们访问登录接口，可能是携带信息不对，导致服务端不认可我们的请求，返回了错误的信息")  
        print(re.text)  
        return None  
  
    data = dist["data"]  
    return data  
  
  
def getMenu(auth):  
    session = requests.session()  
  
    url = "http://47.108.197.167:8008/documentMenu/find/listMenu"  
  
    payload = {  
        "pageSize": 30,  
        "pageNumber": 1  
    }  
  
    if auth == None:  
        print("这里没有获取到jwt，停止执行")  
        return None  
  
    heads = {  
        "authorization": auth  
    }  
  
    re = session.post(url, json=payload, headers=heads, timeout=2)  
  
    # 状态码，请求的状态信息，只有200，就是成功，如果是302，说明是要浏览器进行跳转，如果是其他非200或者302的，都是错误  
    if re.status_code != 200:  
        print("访问服务端失败")  
        return None  
    # re.content,用来获取二进制流，text获取文本信息  
    return json.loads(re.text)  
  
def checkdata(res):  
    if not res:  
        print("前面的流程失败了，无法清洗数据")  
        return None  
    if res['code'] != 200:  
        print("状态码不正确，服务器认为是错误请求")  
        return None  
    data = res['data']['data']  
    list = []  
    for i in data:  
        list.append(i['id'])  
        sonlist = i['list']  
        if not sonlist:  
            continue  
        for j in sonlist:  
            list.append(j['id'])  
            sonsonlist = j['list']  
            if not sonsonlist:  
                continue  
            for k in sonsonlist:  
                list.append(k['id'])  
    return list  
  
def get_Docinfo(list, auth):  
    seesion = requests.Session()  
    url = "http://47.108.197.167:8008/documentMenu/find/dataInfo"  
    heads = {  
        "authorization": auth  
    }  
    for i in list:  
        payload = {"menuId": i}  
        re = seesion.post(url, json=payload, headers=heads, timeout=2)  
        re = json.loads(re.text)  
        if re["code"] != 200:  
            print("这次请求失败")  
            print(re)  
            continue  
        if not re["data"]:  
            continue  
        text = re["data"]["textInfo"]  
        writeFile(text)  
  
  
def writeFile(text):  
    with open("./" + "result.html", "a",encoding="utf-8") as f:  
        f.write(text)  
  
  
#登录  
auth = login()  
# 获取菜单  
res = getMenu(auth)  
#清洗数据  
list = checkdata(res)  
#取得真实数据  
get_Docinfo(list, auth)
